⑤SQLインジェクション対策（検索編のコード）

#２学期前半
検索を実行する際にprepare()とbindValue()を間にかませる。
これで入力データが「単なる文字列」として処理される。
serch_pokemon.php 
// 名前をGETメソッドでリクエスト
$search_name = $_GET['name'] ?? '';   
// 検索結果を入れる配列
$pokemons = [];

// 入力がある場合のみ検索を実行
if ($search_name !== '') {
    // SQLの準備 (:name は後で変数を安全に入れるためのプレースホルダー)
    $stmt = $db->prepare("SELECT * FROM pokemon WHERE name LIKE :name ");
    // 変数を入れる前に、前後に % をくっつける（PHP側で文字列結合）
    $like_search_name = '%' . $search_name . '%';  
    // 変数の割り当て (セキュリティ対策)
    $stmt->bindValue(':name', $like_search_name, PDO::PARAM_STR);
    // 実行
    $stmt->execute();
    // 結果をすべて取得
    $pokemons = $stmt->fetchAll(PDO::FETCH_ASSOC);
}
?>
